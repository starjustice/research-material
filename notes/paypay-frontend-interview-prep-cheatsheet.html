The JavaScript traps that cost points

// 1 — sort without a comparator sorts as TEXT
[10, 9, 1].sort();               // [1, 10, 9]  wrong
[10, 9, 1].sort((a, b) => a - b); // [1, 9, 10] right

// 2 — includes() inside a loop is a hidden O(n^2)
const bSet = new Set(b);
for (const x of a) if (bSet.has(x)) out.push(x);

// 3 — the 2D array reference trap
const bad  = Array(3).fill(Array(3).fill(0));                 // 3 refs to ONE row
const good = Array.from({ length: 3 }, () => new Array(3).fill(0));

// 4 — shift() is O(n). BFS queue with an index pointer:
const q = [start]; let head = 0;
while (head < q.length) { const cur = q[head++]; /* push neighbours */ }

// 5 — strings are immutable. Build with an array.
const parts = []; for (const c of s) parts.push(f(c));
const out = parts.join("");

// 6 — map(Number), never map(parseInt)
["1","2","3"].map(parseInt); // [1, NaN, NaN]
["1","2","3"].map(Number);   // [1, 2, 3]

// 7 — reading stdin, if the platform is Codevue / HackerRank style
const input = require("fs").readFileSync(0, "utf8").trim().split("\n");
const nums  = input[1].split(" ").map(Number);
