// 1 · Debounce — the closure story in 5 lines (live search box)
function debounce<T extends (...a: any[]) => void>(fn: T, ms: number) {
  let timer: ReturnType<typeof setTimeout>;   // lives in the closure
  return (...a: Parameters<T>) => {
    clearTimeout(timer);                       // each keystroke resets the wait
    timer = setTimeout(() => fn(...a), ms);
  };
}
Say: "Debounce for search — I want the final query. Throttle for scroll and resize — regular updates while events keep firing. The timer lives in a closure, which is my go-to practical closure example."
// 2 · The stale closure bug — and the fix
useEffect(() => {
  const id = setInterval(() => {
    setCount(count + 1);        // BUG: closes over first render's count (0)
    setCount(c => c + 1);       // FIX: functional update reads latest value
  }, 1000);
  return () => clearInterval(id);
}, []);                          // empty deps = the closure never refreshes
Say: "Every render's functions close over that render's values. An effect with empty deps keeps the first render's backpack forever. Fixes: functional updates, correct deps, or useEffectEvent in React 19.2."
// 3 · useFetch — loading / error / abort, in one hook
function useFetch<T>(url: string) {
  const [data, setData] = useState<T | null>(null);
  const [error, setError] = useState<Error | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const controller = new AbortController();
    setLoading(true); setError(null);
    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`); // fetch won't reject on 500!
        return res.json() as Promise<T>;
      })
      .then(json => { setData(json); setLoading(false); })
      .catch(err => {
        if (err.name === "AbortError") return;  // cancelled, not an error
        setError(err); setLoading(false);
      });
    return () => controller.abort();  // fixes race on fast url changes + unmount
  }, [url]);

  return { data, error, loading };
}
Close with: "In production I use React Query — it adds caching, deduplication, retries, and stale-while-revalidate. This hook is what it does at its core."
